### 🐛 Cannot export knn_graph

I'm trying to export a model for use in cpp. would like to use onnxruntime ideally but can work with libtorch as well.
How do we export knn_graph?

```
import torch
from torch import nn
from torch_geometric.nn import ChebConv, knn_graph

class SampleModule(nn.Module):
    def __init__(self, in_channels:int, out_channels:int, K:int, aggr:str):
        super().__init__()
        self.conv1 : nn.Module = ChebConv(in_channels, out_channels, K=K, aggr=aggr)

    def forward(self, x:torch.Tensor, position:torch.Tensor) -> torch.Tensor:
        edges = knn_graph(position.squeeze(0), k=20, flow='target_to_source')
        x = self.conv1(x.transpose(2,1), edges)
        return x.transpose(2,1)


m = SampleModule(512, 256, 2, 'add')
m.eval()

model = torch.export.export(m, args=(torch.randn(10, 10), torch.randn(10, 10)), strict=False)
torch.export.save(model, "model.pt2")
```
