```
import os
import gc
import psutil
import torch
import torch_geometric

class CustomDataset(torch_geometric.data.Dataset):
  def __init__(self, root, num_graphs = 150):
    self.num_graphs = num_graphs
    super().__init__(root)
      
  def len(self):
    return self.num_graphs

  @property
  def processed_file_names(self):
    return [f'{idx}.pt' for idx in range(self.num_graphs)]

  def process(self):
    for idx in range(self.num_graphs):
      x = torch.randn((500, 1024))
      edge_index = torch.randint(0, 500, (2, 1500))
      y = torch.tensor([idx % 2], dtype=torch.long)
      data = torch_geometric.data.Data(x = x, edge_index = edge_index, y = y)        
      path = os.path.join(self.processed_dir, f'{idx}.pt')
      torch.save(data, path)

  def get(self, idx):
    path = os.path.join(self.processed_dir, f'{idx}.pt')
    data = torch.load(path, weights_only = False)
    return data

dataset = CustomDataset(root = 'data')
loader = torch_geometric.loader.DataLoader(dataset, batch_size=16, shuffle=True, num_workers = 0)
```
