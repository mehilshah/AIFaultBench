### Reproduction

Between versions `5.8.0` and `5.9.0` Sam3 started raising a `RuntimeError` when compiled using `dtype=torch.bfloat16`. I have also ran on the latest main and the issue is still present in commit 0bc355418bb265136a66c2dedc501066ffbc237d

This script reproduces the error,

```python
# /// script
# requires-python = ">=3.10"
# dependencies = [
#     "transformers",
#     "torch",
#     "torchvision",
#     "pillow",
# ]
# ///
import torch
import transformers
from PIL import Image
from transformers import Sam3Model, Sam3Processor

print("torch       ", torch.__version__)
print("transformers", transformers.__version__)

model_id = "facebook/sam3"
device = "cuda"
processor = Sam3Processor.from_pretrained(model_id)
model = (
    Sam3Model.from_pretrained(model_id, attn_implementation="sdpa", dtype=torch.bfloat16)
    .eval()
    .to(device)
)

image = Image.new("RGB", (512, 512), (128, 128, 128))
inputs = processor(images=image, text="cat", return_tensors="pt").to(device)

# Eager succeeds
with torch.inference_mode():
	outputs = model(**inputs)
print("eager OK:", type(outputs).__name__)

# Compiled raises type mismatch error
compiled = torch.compile(model)
with torch.inference_mode():
    outputs = compiled(**inputs)
print("compiled OK:", type(outputs).__name__)
```
