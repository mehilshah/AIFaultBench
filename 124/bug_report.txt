Title: [Bug]: putting `Sequence` space inside `Dict`, `Tuple`, or `OneOf` space will fail the env checker
Date: 2025-08-29T20:13:33+00:00
URL: https://github.com/DLR-RM/stable-baselines3/issues/2172

Description:
### 🐛 Bug

`Sequence` space is not properly handled when putting inside a composite space (`Dict`, `Tuple`, or `OneOf`), causing the `stable_baselines3.common.env_checker.check_env` to fail.

### Code example

```python
import traceback
from itertools import product
from typing import Optional

import gymnasium as gym
import numpy as np
from gymnasium import spaces
from stable_baselines3.common.env_checker import check_env


class DummyDictEnv(gym.Env):
    metadata = {"render_modes": ["human"]}

    def __init__(self, stack: bool, inside: str):
        self.observation_space = spaces.Sequence(
            spaces.Box(low=-100, high=100, shape=(1,), dtype=np.float32),
            stack=stack,
        )
        if inside == "dict":
            self.observation_space = spaces.Dict({"seq": self.observation_space})
        elif inside == "tuple":
            self.observation_space = spaces.Tuple((self.observation_space,))
        elif inside == "oneof":
            self.observation_space = spaces.OneOf((self.observation_space, spaces.Discrete(3)))
        self.action_space = spaces.Discrete(2)

    def seed(self, seed=None):
        if seed is not None:
            self.observation_space.seed(seed)

    def step(self, action):
        reward = 0.0
        terminated = truncated = False
        return self.observation_space.sample(), reward, terminated, truncated, {}

    def reset(self, *, seed: Optional[int] = None, options: Optional[dict] = None):
        if seed is not None:
            self.observation_space.seed(seed)
        return self.observation_space.sample(), {}

    def render(self):
        pass


for inside, stack in product(("none", "dict", "tuple", "oneof"), (False, True)):
    print(f"{inside=} {stack=}:")
    try:
        check_env(DummyDictEnv(stack=stack, inside=inside))
        print("PASSED")
    except Exception:
        traceback.print_exc()
    print("=" * 80)
```
