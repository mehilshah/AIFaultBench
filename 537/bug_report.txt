def get_data_loader (root, batch_size, num_workers, distributed, args , rank):
    train_transform = torchvision.transforms.Compose([
        transforms.RandomCrop (32, padding = 4),
        transforms.RandomHorizontalFlip (),
        transforms.RandomRotation (15),
        transforms.ToTensor (),
        transforms.Normalize (mean, std)
    ])
    test_transform = torchvision.transforms.Compose([
        transforms.ToTensor (),
        transforms.Normalize (mean, std)
    ])
    train_set = torchvision.datasets.CIFAR100 (root = root, train = True, download = True,
                                               transform = train_transform)
    test_set = torchvision.datasets.CIFAR100 (root = root, train = False, download = True,
                                               transform = test_transform)

    if  distributed:
        train_sampler = torch.utils.data.distributed.DistributedSampler (train_set,
                                                                        num_replicas=args.world_size,
    	                                                                rank=rank)
        test_sampler = torch.utils.data.distributed.DistributedSampler (test_set,
                                                                        num_replicas = args.world_size,
                                                                        shuffle = False,
                                                                        rank = rank)
        train_loader = torch.utils.data.DataLoader (dataset = train_set,
                                  pin_memory = True,
                                  shuffle = False,
                                  batch_size = batch_size,
                                  num_workers = num_workers,

    else:
        train_loader = torch.utils.data.DataLoader (train_set, batch_size = batch_size, shuffle = True,
                                                    num_workers = num_workers, pin_memory = True,
                                                    pin_memory_device = 'cpu')
        test_loader = torch.utils.data.DataLoader (test_set, batch_size = batch_size, shuffle = False,
                                                    num_workers = num_workers, pin_memory = True,
                                                    pin_memory_device = 'cpu')
    return train_loader, test_loader

def get_model(model_name, model_dir, num_classes, device, distributed, rank):
    model = timm.create_model (model_name, pretrained = False,
                               cache_dir = model_dir, num_classes = num_classes)

    # from models import resnet
    # model = resnet.resnet34()

    model = model.to(device, memory_format = torch.channels_last)

    if distributed:
        model = torch.nn.SyncBatchNorm.convert_sync_batchnorm (model)
        model = torch.nn.parallel.DistributedDataParallel (model, device_ids = [rank])
    model = torch.compile (model, mode = 'default', disable = True)
    for param in model.parameters ():
        param.requires_grad = True
    return model

def train(epochs, train_loader, model, criterion, scaler, optimizer, device, data_loader_test, rank, lr_scheduler):
    best_acc = 0.0
    for epoch in range (1, epochs + 1):
        loss_meter = AverageMeter()
        model.train ()
        train_loader.sampler.set_epoch (epoch)
        for inputs, targets in tqdm (train_loader, desc = f'[Training] Epoch {epoch}/{epochs}',
                                     unit = 'batch',disable = (rank != 0),):
            inputs, targets = (inputs.to (device, non_blocking = True, memory_format = torch.channels_last),
                               targets.to (device, non_blocking = True))
            # with torch.autocast (device_type = "cuda", dtype = torch.float16):
            #     outputs = model (inputs)
            #     loss_value = criterion (outputs, targets)
            outputs = model (inputs)
            loss_value = criterion (outputs, targets)
            loss = reduce_tensor (loss_value)

            loss_meter.update (loss.item (), targets.size (0))
            # scaler.scale (loss_value).backward ()
            # scaler.step (optimizer)
            # scaler.update ()
            # optimizer.zero_grad ()
            loss_value.backward ()
            optimizer.step ()
            optimizer.zero_grad ()
        lr_scheduler.step()
        acc1, loss = validate(data_loader_test, model, device, criterion)
        if rank  == 0:
            if acc1 > best_acc:
                best_acc = acc1
                # torch.save(model.state_dict(), 'best.pth')
            print(f"Accuracy={acc1:.2f}% "
                  f"Loss_train={loss_meter.avg:.3f} "
                  f"Loss_test={loss:.3f} "
                  f"lr={lr_scheduler.get_last_lr().pop():.4f} "
                  f"best_acc={best_acc:.2f}")

def reduce_tensor(tensor):
    rt = tensor.clone()
    dist.all_reduce(rt, op=dist.ReduceOp.SUM)
    rt /= dist.get_world_size()
    return rt

@torch.inference_mode()
def validate(data_loader, model, device, criterion):
    model.eval()

    loss_meter = AverageMeter()
    acc1_meter = AverageMeter()

    for images, targets in tqdm (data_loader, desc = f'[Testing]', unit = 'batch',disable = (rank != 0)):
        images, targets = (images.to(device, non_blocking = True, memory_format = torch.channels_last),
                           targets.to(device, non_blocking = True))

        # compute output
        # with torch.autocast (device_type = "cuda", dtype = torch.float16):
        output = model(images)

        # measure accuracy and record loss
        loss = criterion(output, targets)
        acc1, _ = accuracy(output, targets, topk=(1,5))
        acc1 = reduce_tensor(acc1)
        loss = reduce_tensor(loss)

        loss_meter.update(loss.item(), targets.size(0))
        acc1_meter.update(acc1.item(), targets.size(0))

    return acc1_meter.avg, loss_meter.avg

def main(gpu, args):
    rank = args.nr * args.gpus + gpu
    torch.distributed.init_process_group (backend = 'nccl',init_method='env://',

           optimizer = optimizer, device = device, data_loader_test =  data_loader_test,
           rank = rank, lr_scheduler = lr_scheduler)
