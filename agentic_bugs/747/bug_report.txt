def get_llm() -> ChatOpenAI:
    """
    Get the ChatOpenAI LLM instance.

    Returns:
        BaseChatModel: Configured LLM instance
    """
    global _llm

    if _llm is None:
        _llm = ChatOpenAI(

            base_url=settings.LLM_BASE_URL,
            api_key=settings.LLM_API_KEY,
        )

    return _llm
