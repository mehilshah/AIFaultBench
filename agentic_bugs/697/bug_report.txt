# --- Blocked function with stack info ---
def _blocked(func_name, *args, **kwargs):
    # Get the caller frame
    stack = inspect.stack()
    # stack[1] is the immediate caller
    caller = stack[1]
    filename = caller.filename
    lineno = caller.lineno
    return f"Blocked call to os.{func_name}() from {filename}, line {lineno}"

# Patch dangerous OS functions
for func_name in ["system", "popen", "execl", "execv", "execvp"]:
