TITLE
[BUG] Torch compile slower + NaNs

BODY
**Describe the bug**
I am currently trying to train `tiny_vit_21m_224.dist_in22k_ft_in1k` for a binary classification task. I am getting NaN, and also the compiled model is slower than the eager model.

**To Reproduce**
Steps to reproduce the behavior:
```python
import timm
import torch
from torch import optim
import dali_train_loader

torch.set_float32_matmul_precision("high")
torch.backends.cudnn.benchmark = True

images_dir = "/image_datasets/jpeg_experiment"

def loss_func(pred, y):
    criterion = torch.nn.BCEWithLogitsLoss()
    loss = criterion(pred, y)
    return loss

def load_model(
    model_name: str = "tiny_vit_21m_224.dist_in22k_ft_in1k",
    num_classes=1,
    device: str = "cuda",
):
    device = torch.device(device=device)
    model = timm.create_model(
        model_name=model_name, pretrained=True, num_classes=num_classes
    ).to(device)
    model = torch.compile(model=model, fullgraph=True, backend="inductor")
    return model

model = load_model()

dummy_x = torch.randn(128, 3, 224, 224, device="cuda")
for _ in range(100):
    model(dummy_x)

optimizer = optim.AdamW(model.parameters(), lr=3e-4)

train_data = dali_train_loader(images_dir=images_dir, batch_size=128)

start_event = torch.cuda.Event(enable_timing=True)
end_event = torch.cuda.Event(enable_timing=True)

start_event.record()
model.train()
for epoch in range(1):
    for i, data in enumerate(train_data):
        inputs, labels = data[0]["data"].float(), data[0]["label"].float()

        with torch.amp.autocast(device_type="cuda", dtype=torch.bfloat16):
            pred = model(inputs)
            loss = loss_func(pred, labels)
        loss.backward()
        optimizer.step()
        optimizer.zero_grad()

        if i % 10 == 0:
            print(f"Epoch [{epoch + 1}/10], Batch [{i + 1}], Loss: {loss.item():.4f}")
end_event.record()
torch.cuda.synchronize()
training_time_ms = start_event.elapsed_time(end_event)
training_time_s = training_time_ms / 1000.0
print(f"Training time: {training_time_s} seconds")
```

**Expected behavior**
I would expect the model to not have NaN when training with compile and to also be faster.
