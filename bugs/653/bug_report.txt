### Problem description

When running ChatAgent with AWS Bedrock Converse backend and function tools, CAMEL sends a request sequence that Bedrock rejects with:
```
botocore.errorfactory.ValidationException: An error occurred (ValidationException) when calling the Converse operation: Expected toolResult blocks at messages.2.content for the following Ids: tooluse_bRX9ftpYK3xO5tev2x4qtc
```
Bedrock rejects the request due to missing toolResult blocks for emitted toolUse IDs. The error occurs because Bedrock's API requires that every toolUse in an assistant message must be immediately followed by a user message containing corresponding toolResult blocks.
For example:
```
...
5 assistant ['text', 'toolUse', 'toolUse', 'toolUse'] ['tooluse_9Wpx82Jgfft7cqfo3fb5lp', 'tooluse_XWmAH7SUJCV9KjZqenyC4q', 'tooluse_VWjdWJGHj4sxuDCW0wg80Q']
6 user ['toolResult'] ['tooluse_9Wpx82Jgfft7cqfo3fb5lp']
7 user ['toolResult'] ['tooluse_XWmAH7SUJCV9KjZqenyC4q']
8 user ['toolResult'] ['tooluse_VWjdWJGHj4sxuDCW0wg80Q']
...
```
After one assistant message with 3 toolUses, the payload sends 3 separate user toolResult messages, but Converse expects those pending toolResult blocks grouped in the immediate next user message.

The Python snippets:

```python
import os
from pathlib import Path
from typing import Any, Dict

import boto3
from botocore.config import Config
from dotenv import load_dotenv

from camel.agents import ChatAgent
from camel.messages import BaseMessage
from camel.models import ModelFactory
from camel.toolkits import FunctionTool
from camel.types import ModelPlatformType


def tool_a(topic: str) -> Dict[str, Any]:
    """Return a deterministic value for tool A."""
    return {"tool": "tool_a", "topic": topic, "value": "result_a"}


def tool_b(topic: str) -> Dict[str, Any]:
    """Return a deterministic value for tool B."""
    return {"tool": "tool_b", "topic": topic, "value": "result_b"}


def build_bedrock_client() -> Any:
    gateway_url = os.getenv("AI_GATEWAY_URL") or os.getenv("BEDROCK_ENDPOINT_URL")
    api_key = os.getenv("AI_GATEWAY_KEY") or os.getenv("AWS_BEARER_TOKEN_BEDROCK")

    if not gateway_url or not api_key:
        load_dotenv(dotenv_path=Path(".env"))
        gateway_url = gateway_url or os.getenv("AI_GATEWAY_URL") or os.getenv("BEDROCK_ENDPOINT_URL")
        api_key = api_key or os.getenv("AI_GATEWAY_KEY") or os.getenv("AWS_BEARER_TOKEN_BEDROCK")

    if not gateway_url or not api_key:
        raise ValueError("Missing gateway URL/API key. Set AI_GATEWAY_URL and AI_GATEWAY_KEY in env or .env")

    endpoint = gateway_url if gateway_url.endswith("/bedrock") else f"{gateway_url}/bedrock"
    os.environ["AWS_BEARER_TOKEN_BEDROCK"] = api_key

    return boto3.client(
        service_name="bedrock-runtime",
        region_name="us-east-1",
        endpoint_url=endpoint,
        aws_access_key_id="",
        aws_secret_access_key="",
        config=Config(retries={"max_attempts": 0}),
    )


def main() -> None:
    model_id = os.getenv("BEDROCK_MODEL_ID", "amazon.nova-lite-v1:0")
    client = build_bedrock_client()

    model = ModelFactory.create(
        model_platform=ModelPlatformType.AWS_BEDROCK_CONVERSE,
        model_type=model_id,
        model_config_dict={
            "temperature": 0.0,
            "max_tokens": 256,
        },
        bedrock_client=client,
    )

    tools = [FunctionTool(tool_a), FunctionTool(tool_b)]
    agent = ChatAgent(
        system_message="You are a test assistant.",
        model=model,
        tools=tools,
        retry_attempts=1,
        step_timeout=120,
    )

    user_msg = BaseMessage.make_user_message(
        role_name="user",
        content=(
            "Call tool_a and tool_b in the same assistant turn. "
            "Then summarize both tool results in one short sentence."
        ),
    )

    print("Running ChatAgent repro with two dummy tools...")
    print(f"Model: {model_id}")
    
    response = agent.step(user_msg)

    for i, msg in enumerate(response.msgs):
        print(f"msg[{i}] role={msg.role_name}")
        print(msg.content[:400])

if __name__ == "__main__":
    main()
```
