# recovered-from-issue: https://github.com/deepspeedai/DeepSpeed/issues/7686

TITLE
[BUG] Zenflow_stage3 - RuntimeError: narrow() cannot be applied to a 0-dim tensor.

BODY
**Describe the bug**
When running ZeRO Stage 3 with the ZenFlow optimizer, the training fails during the backward pass on the first optimization step (`model_engine.backward(loss)`).

The crash is caused by a `RuntimeError: narrow() cannot be applied to a 0-dim tensor` originating inside the `group_step` function of the class `ZenFlowSelectiveAdamW_stage3`.

Detailed debugging revealed a corruption in the internal DeepSpeed parameter state: The parameter's metadata (`param.ds_shape`) correctly reports a 2D tensor (e.g., `torch.Size([3072, 1024])`), but the parameter's underlying data handle (`param.ds_tensor.data`) is incorrectly initialized as a 0-dimensional scalar (`torch.Size([])`), leading to the crash when the narrow() method is called on it.

**To Reproduce**
Steps to reproduce the behavior:
1. Configure a DeepSpeed JSON file (`ds_config_zf_shell.json`) to enable ZeRO Stage 3 with Optimizer Offload and the ZenFlow extension.
2. Launch the finetune_zf.py script using the DeepSpeed launcher.

**finetune_zf.py**
```python
import torch
import time
import deepspeed
import argparse
import os
import random
import numpy as np
import datetime
from datasets import load_dataset
from datasets.formatting.formatting import LazyBatch
from torch.utils.data import DataLoader
from transformers import (
    AutoModelForCausalLM,
    AutoTokenizer,
    DataCollatorForLanguageModeling,
)
from deepspeed import comm as dist

os.environ["TOKENIZERS_PARALLELISM"] = "false"

def set_seed(seed):
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)

def format_chat_template(question: str, answer: str, tokenizer) -> str:
    messages = [
        {'role': 'user', 'content': question},
        {'role': 'assistant', 'content': answer},
    ]
    
    chat_template = tokenizer.apply_chat_template(
        messages,
        tokenize=False,
        add_generation_prompt=False,
    )
    return chat_template

def tokenize_function(examples: LazyBatch, tokenizer, max_length: int) -> torch.Tensor:
    chat_templates = [
        format_chat_template(question, answer, tokenizer)
        for question, answer in zip(examples['prompt'], examples['response'])
    ]
    
    return tokenizer(
        chat_templates, 
        truncation=True, 
        padding='max_length', 
        max_length=max_length, 
        return_tensors='pt'
    )


def log_event(message: str):
    if dist.get_rank() == 0:
        # Get current time with microseconds
        now = datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S.%f')
        print(f"[{now}] {message}", flush=True) # flush=True ensures it prints immediately

def main(args):    
    # Initialize DeepSpeed
    # This will automatically handle setting the device (e.g., args.local_rank)
    deepspeed.init_distributed()
    
    set_seed(args.seed)

    # Load Tokenizer
    tokenizer = AutoTokenizer.from_pretrained(args.model_path)
    if tokenizer.pad_token is None:
        tokenizer.pad_token = tokenizer.eos_token
        
    if dist.get_rank() == 0:
        print("Tokenizer loaded.")

    # Load Model
    model = AutoModelForCausalLM.from_pretrained(
        args.model_path,
    )
    
    # Resize token embeddings if tokenizer has new tokens
    model.resize_token_embeddings(len(tokenizer))
    
    # --- End of Fix ---
    if dist.get_rank() == 0:
        print(f"Model loaded. Embedding size: {model.get_input_embeddings().weight.shape}")

    # Load and Process Dataset
    dataset = load_dataset('Dahoas/rm-static', split='train')
    # Use a smaller subset for quick testing if needed
    # samples = dataset.select(range(3000))
    
    tokenized_dataset = dataset.map(
        tokenize_function,
        batched=True,
        # Pass tokenizer and max_length to the map function
        fn_kwargs={'tokenizer': tokenizer, 'max_length': args.max_seq_len},
        remove_columns=dataset.column_names # Keep only tokenized columns
    )
    
    if dist.get_rank() == 0:
        print("Dataset processed.")

    # Data Collator
    data_collator = DataCollatorForLanguageModeling(tokenizer=tokenizer, mlm=False)

    # DeepSpeed Initialization
    model_engine, optimizer, train_dataloader, lr_scheduler = deepspeed.initialize(
        args=args,
        model=model,
        model_parameters=model.parameters(),
        training_data=tokenized_dataset,
        collate_fn=data_collator
    )
    
    if dist.get_rank() == 0:
        print("DeepSpeed engine initialized.")

    # --- Training Loop ---
    model_engine.train()
    global_step = 0
    
    if dist.get_rank() == 0:
        print("Starting training...")

    for epoch in range(args.num_train_epochs):
        if dist.get_rank() == 0:
            print(f"--- Starting Epoch {epoch + 1}/{args.num_train_epochs} ---")
            
        for step, batch in enumerate(train_dataloader):
            log_event(f"[Step {global_step + 1}][start]")
            step_start_time = time.time()
            
            # Move batch to the correct device
            batch = {k: v.to(model_engine.device) for k, v in batch.items()}
            
            # Forward pass
            log_event("[Forward][start]")
            outputs = model_engine(**batch)
            loss = outputs.loss

            # --- 2. Backward Pass ---
            log_event("[Backward][start]")
            # Backward pass
            model_engine.backward(loss)

            log_event("[Update][start]")
            # Optimizer step
            model_engine.step()

            step_time = time.time() - step_start_time
            global_step += 1
            
            # Log to console
            if dist.get_rank() == 0 and global_step % args.log_steps == 0:
                print(f"Step {global_step:5d} | Loss: {loss.item():.4f} | Time: {step_time*1000:.0f}ms | LR: {lr_scheduler.get_last_lr()[0]:.2e}")
            log_event(f"[Step {global_step}][end]")
            # Check if max_steps is reached
            if args.max_steps > 0 and global_step >= args.max_steps:
                if dist.get_rank() == 0:
                    print(f"Reached max_steps ({args.max_steps}). Stopping training.")
                break
        
        if args.max_steps > 0 and global_step >= args.max_steps:
            break

        log_event(f"[Epoch {epoch + 1}][end]")


    # --- End of Training ---

    # Save checkpoint
    if dist.get_rank() == 0:
        print("Training finished. Saving checkpoint...")
        model_engine.save_checkpoint(args.output_dir)
        # Also save the tokenizer for easy-to-use-later
        tokenizer.save_pretrained(args.output_dir)
        print(f"Model and tokenizer saved to {args.output_dir}")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="DeepSpeed Core Training Script (No Trainer)")
    
    # --- Script-specific arguments ---
    parser.add_argument("--model_path", type=str, required=True,
                        help="Path to the pretrained model (e.g., /usr/local/models/Llama-3.2-3B-Instruct)")
    parser.add_argument("--output_dir", type=str, required=True,
                        help="Directory to save checkpoints and tokenizer")
    parser.add_argument("--max_seq_len", type=int, default=1024,
                        help="Maximum sequence length for tokenization")
    parser.add_argument("--num_train_epochs", type=int, default=1,
                        help="Number of training epochs")
    parser.add_argument("--max_steps", type=int, default=-1,
                        help="Total number of training steps. Overrides num_train_epochs. (-1 to ignore)")
    parser.add_argument("--log_steps", type=int, default=1,
                        help="Log training status every X steps")
    parser.add_argument("--seed", type=int, default=42,
                        help="Random seed for reproducibility")
    
    # --- DeepSpeed arguments ---
    # This argument is required by deepspeed.initialize()
    # It tells DeepSpeed to find its config (e.g., --deepspeed ds_config.json)
    parser.add_argument('--local_rank',
                        type=int,
                        default=-1,
                        help='local rank passed from distributed launcher')
    
    # Add all of DeepSpeed's arguments
    parser = deepspeed.add_config_arguments(parser)
    
    args = parser.parse_args()

    main(args)
```
